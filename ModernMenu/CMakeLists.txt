cmake_minimum_required(VERSION 3.18)
project(RightClickOfficeToPDF LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(ExplorerCommand SHARED ExplorerCommand.cpp ExplorerCommand.def Launch.cpp)
target_link_libraries(ExplorerCommand PRIVATE ole32 shlwapi)
add_executable(OfficeToPDF WIN32 App.cpp Launch.cpp)
target_link_libraries(OfficeToPDF PRIVATE ole32 shell32)
install(TARGETS ExplorerCommand OfficeToPDF RUNTIME DESTINATION .)
